    static void HarlequinRotates(int harlequinID){
        int dir = Player.GetDirection();

        if (dir == 0){
            HarlequinRotatesFacingSouth(harlequinID);
        }
        else if (dir == 1){
            HarlequinRotatesFacingNorth(harlequinID);
        }
        else if (dir == 2){
            HarlequinRotatesFacingEast(harlequinID);
        }
        else {
            HarlequinRotatesFacingWest(harlequinID);
        }
        Actor.WaitActionFinish();
    }
