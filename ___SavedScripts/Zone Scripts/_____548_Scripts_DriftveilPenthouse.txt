    static void JanineAmbush(){
        int playerX; 
        int playerZ; 
        Player.GetGPos(playerX, playerZ);
        int dir = Player.GetDirection();
        if (dir == 0){
            Actor.SetGPos(5, playerX, 0, playerZ+1, dir);
        }
        else if (dir == 1){
            Actor.SetGPos(5, playerX, 0, playerZ-1, dir);
        }
        else if (dir == 2){
            Actor.SetGPos(5, playerX+1, 0, playerZ, dir);
        }
        else {
            Actor.SetGPos(5, playerX-1, 0, playerZ, dir);
        }
        ShadowAppear(5);
        
    }
    
    static void PlayerBacksUpFromJanine(){
        int dir = Player.GetDirection();
        if (dir == 0){
            PlayerBacksUpSouth(255);
        }
        else if (dir == 1){
            PlayerBacksUpNorth(255);
        }
        else if (dir == 2){
            PlayerBacksUpWest(255);
        }
        else {
            PlayerBacksUpEast(255);
        }
    }
