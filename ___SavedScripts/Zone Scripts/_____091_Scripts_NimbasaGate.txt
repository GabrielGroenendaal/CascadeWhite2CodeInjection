	public static void main_5() {
		int playerX; 
		int playerZ;
		int isVictory; 
		
		Actor.PauseAll();
		Sound.SEPlay(1351);
		Message.Info(3, 2);
		Message.WaitAdvance();
		Message.CloseAll();
		EventFlags.Clear(1712);
		Actor.Show(3);
		ShadowAppear(3);
		Actor.WaitActionFinish();
		TurnEast8F(255);
		ShowExclamationMark(255);
		Actor.WaitActionFinish();
		Player.GetGPos(playerX, playerZ);
		Sound.BGMPlay(1239);
		if (playerZ != 5) {
			Actor.WalkRoute(3, 7, playerZ, 1, 4, 1);
		}
		WalkWestFast(3);
		Actor.WaitActionFinish();

		Message.ActorEx(1024, 4, 3, 0, 0); 
		Message.CloseAll();
		Message.System(7, 2);
		Message.CloseAll();
		Battle.CallTrainerBattle(236, 0, 0);
		isVictory =  Battle.IsTrainerVictory();
		if (isVictory != 1) {
			Battle.CallTrainerLose();
			goto LABEL_ENDAMBUSH;
		}
		Battle.EndTrainerBattle();
		BackUpSlowlyEast(3);
		Actor.WaitActionFinish();
		Message.ActorEx(1024, 5, 3, 0, 0); 
		Message.CloseAll();
		ShadowDisappear(3);
		Actor.WaitActionFinish();
		Actor.Hide(3);
		EventFlags.Set(1712);
		EventWorks.Set(16533, 3);

		LABEL_ENDAMBUSH:
		Runtime.FinishSubEvents();
		Actor.UnpauseAll();
	}
