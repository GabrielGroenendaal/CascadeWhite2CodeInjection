	static int checkForUnown(){
		int UnownIdx;
		int isUnown;

		int count = 0;
		LABEL_COUNT:
		if (count < 24){
			PokeParty.FindPkmEx(201, count, UnownIdx, isUnown);

			if (isUnown == 1){
				return UnownIdx;
			}
			count = count + 1;

		}

	}
