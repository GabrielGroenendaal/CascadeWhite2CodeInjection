    public static void main_8(){
    	int gruntTotal;
        if (!EventFlags.Get(348)){
            gruntTotal += 1;
        }
        if (!EventFlags.Get(349)){
            gruntTotal += 1;
        }
        if (!EventFlags.Get(350)){
            gruntTotal += 1;
        }
        if (!EventFlags.Get(353)){
            gruntTotal += 1;
        }
        if (gruntTotal == 0){
            int gruntX; 
            int gruntZ;
            int playerX;
            int playerZ; 

            //Actor.PauseAll();

            // Reposition the Grunt 
            Actor.GetGPos(0, gruntX, gruntZ);
            Player.GetGPos(playerX, playerZ);
			ShowExclamationMark(0);
			if (playerZ>=43){
				TurnSouth8F(0);
			}
			else if(playerZ<=38){
				TurnNorth8F(0);
			}
			else {
				if (playerX > gruntX){
				TurnEast8F(0);
				Actor.WaitActionFinish();
				}
				else {
					TurnWest8F(0);
					Actor.WaitActionFinish();
				}
			}
			Message.ActorEx(1024, 28, 0, 3, 0);
			Message.CloseAll();
			
            if (gruntZ != playerZ){
                Actor.WalkRoute(0, gruntX, playerZ, 0, 8, 0);
            }
            Actor.WaitActionFinish();
            if (gruntX > playerX){
                Actor.WalkRoute(0, playerX+1, playerZ, 0, 8, 0);
				TurnEast8F(255);
            }
            else {
                Actor.WalkRoute(0, playerX-1, playerZ, 0, 8, 0);
				TurnWest8F(255);
            }
            Actor.WaitActionFinish();

            Message.ActorEx(1024, 18, 0, 3, 0);
            Message.CloseAll();
            Global2805(633, 1);
			EventFlags.Set(356);
            Message.ActorEx(1024, 19, 0, 3, 0);
            if (EventWorks.Get(16512) == 0){
                Message.ActorEx(1024, 20, 0, 3, 0);
                Message.CloseAll();
            }
            else {
                Message.ActorEx(1024, 26, 0, 3, 0);
                EventFlags.Set(572);
                Message.CloseAll();
            }
            EventWorks.Set(16513, 1);
            Runtime.FinishSubEvents();
		    //Actor.UnpauseAll();
        }
    }
