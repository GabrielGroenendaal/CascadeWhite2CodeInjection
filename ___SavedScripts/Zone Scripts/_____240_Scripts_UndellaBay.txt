	public static void main_1() { // @ 0x2c
		int v35;
		int v36;

		v35 = RTC.GetDayOfWeek();
		v36 = Game.GetVersion();
		if (v36 != 23 || v35 != 1 || EventFlags.Get(2775)) {
			if (v36 != 22 || v35 != 4 || EventFlags.Get(2775)) {
				goto LABEL_00ba;
			}
			EventFlags.Clear(903);
			EventWorks.Set(16416, 324);
			return;
		}
		EventFlags.Clear(903);
		EventWorks.Set(16416, 323);
		return;

		LABEL_00ba:
		EventFlags.Set(903);
	}

    // Triggering the Swimmer to Battle you!
    public static void main_7() {
        int yesNo;
		int isVictory;
		int SwimmerEntity;
		int swimmerX;
		int swimmerY;
		int playerX; 
		int playerY;

		SwimmerEntity = 3; // CHANGE THIS LATER
		Actor.PauseAll();

        Player.GetGPos(playerX, playerY);
		Actor.GetGPos(SwimmerEntity, swimmerX, swimmerY);

        Message.Spiked(31, 2);
		ShowExclamationMark(255);
		Actor.WaitActionFinish();
		Message.WaitAdvance();
        Message.CloseAll();
	
        if (playerX > swimmerX){
            Fade.OutBlackQ();
		  Fade.Wait();
		  TurnWest8F(255);
		  Actor.WaitActionFinish();
            Actor.SetGPos(SwimmerEntity, (playerX-1), -6, playerY, 3);
            Runtime.Sleep(32);
            Fade.InBlackQ();
		  Fade.Wait();
        } 
        else {
            Fade.OutBlackQ();
		  Fade.Wait();
		  TurnEast8F(255);
		  Actor.WaitActionFinish();
            Actor.SetGPos(SwimmerEntity, (playerX+1), -6, playerY, 2);
            Runtime.Sleep(32);
            Fade.InBlackQ();
		  Fade.Wait();
        }
        
		Message.ActorEx(1024, 27, SwimmerEntity, 0, 0);
		Message.CloseAll();
		Message.System(36, 2);
        yesNo = YesNoWin.Show();
		if (yesNo != 0) {
			Message.CloseAll();
			Message.ActorEx(1024, 30, SwimmerEntity, 0, 0);
			Input.LastKeyWait();
			Message.CloseAll();

			if (playerX > swimmerX){
                SwimEast3Tiles(255);
            }
            else {
                SwimWest3Tiles(255);
            }
			Actor.WaitActionFinish();
			goto Label_EndAbyssalRuinsGatekeeper;
		}
        Message.CloseAll();
		Message.ActorEx(1024, 28, SwimmerEntity, 0, 0);
		Input.LastKeyWait();
		Message.CloseAll();
		Battle.CallTrainerBattle(315, 0, 0);
        isVictory = Battle.IsTrainerVictory();
        if (isVictory != 1) {
			Battle.CallTrainerLose();
		}
		Battle.EndTrainerBattle();
		Message.ActorEx(1024, 29, SwimmerEntity, 0, 0);
        Input.LastKeyWait();
        Message.CloseAll();
		if (playerX > swimmerX){
            SwimWest3Tiles(3);
        }
        else {
            SwimEast3Tiles(3);
        }
        Actor.WaitActionFinish();
		EventWorks.Set(16504, 2);


		Label_EndAbyssalRuinsGatekeeper:
		Runtime.FinishSubEvents();
		Actor.UnpauseAll();
    }
