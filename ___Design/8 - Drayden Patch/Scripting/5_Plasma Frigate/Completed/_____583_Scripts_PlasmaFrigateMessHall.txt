    // Scientist 
    public static void main_8(){
        int SCIENTIST_ID;
        int isVictory; 
        int yesNo; 

        Actor.PauseAll();
		Sound.SEPlay(1351);
		Actor.FacePlayer();

        if (EventFlags.Get(1520 + SCIENTIST_ID)){
            Message.Actor(1024, 15, 0, 0);
            yesNo = YesNoWin.Show();
            if (yesNo == 1){
                Message.Actor(1024, 16, 0, 0);
                Message.CloseAll();
            }
            else {
                Message.Actor(1024, 17, 0, 0);
                Message.CloseAll();

                Battle.CallTrainerBattle(SCIENTIST_ID, 0, 0);
                isVictory = Battle.IsTrainerVictory();
                if (isVictory != 1){
                    Battle.CallTrainerLose();
                }
                Battle.EndTrainerBattle();
                Message.Actor(1024, 18, 0, 0);
                Message.CloseAll();
                EventFlags.Set(1520 + SCIENTIST_ID);
                Global2805(1, 1); // CHANGE THIS LATER
            }
        }
        else {
            Message.Actor(1024, 19, 0, 0);
            Message.CloseAll();
        }
		
		Runtime.FinishSubEvents();
		Actor.UnpauseAll();

    }

    static void battleOutcome() { // @ 0x3ec
		int v8;
		int v16;

		v16 = Battle.IsTrainerVictory();
		if (v16 != 1) {
			Battle.CallTrainerLose();
			return;
		}
		v8 = PokeParty.GetPkmCount(2);
		if (v8 != 0) {
			goto LABEL_041e;
		}
		PokeParty.RecoverAll();

		LABEL_041e:
		Battle.EndTrainerBattle();
	}
