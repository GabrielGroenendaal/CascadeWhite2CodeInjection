static int callWildBattle(int pokeId, int level, int forme){

}

static int sub_Starters() {

    int listMenu; 
    int yesNo; 
    // Which starter would you like to choose?
    Message.System(textId + 13, 2);

    ListMenu.InitRightAnchor(31, 5, 0, 1, listMenu);
    ListMenu.AddItem(textId+14, 65535, 498); // Tepig
    ListMenu.AddItem(textId+15, 65535, 495); // Snivy
    ListMenu.AddItem(textId+16, 65535, 501); // Oshawott
    ListMenu.AddItem(textId+6, 65535, 3); // Exit 
    ListMenu.ShowDialog();
Label_RepeatStarters: ;
    if (listMenu == 498){
        wildOutcome = callWildBattle(498, 5, 0);
    }
    else if (listMenu == 495){
        wildOutcome = callWildBattle(495, 5, 0);
    }
    else if (listMenu == 501){
        wildOutcome = callWildBattle(501, 5, 0);
    }
    else {
        Message.CloseAll();
        Return 0; 
    }
    if (wildOutcome == 0){
        goto Label_RepeatStarters;
    } else {
        // Ready to move onto the next encounter?
        Message.System(textId + 18, 2);
        yesNo = YesNoWin.Show();
        if (yesNo == 0){
            if (listMenu == 495){
                EventWorks.Set(16432, 0);
            }
            else if (listMenu == 498){
                EventWorks.Set(16432, 1);
            }else {
                EventWorks.Set(16432, 2);
            }
            return 0;
        } else {
            return 1;
        }
    }
}

        if (zoneId == ROUTE7){  // CHANGE THIS LATER
            return 3;
        }
        if (zoneId === 370){
            return 3;
        }
    }
    if (speciesId == 422 || speciesId == 423){
        if (zoneId = 331 || zoneId == 463){
            return 1;
        }
    }

    if (speciesId == 586){
        if (zoneId == ROUTE7 || zoneId == 370){   // ChANGE THIS LATER
            return 3;
        }
        if (zoneId == DEEPLOSTLORN){ // ChANGE THIS LATER
            return 2;
        }
        if (zoneId == 331){
            return 1;
        }
    }

    if (speciesId == 413){
        if (zoneId == 515){
            return 1;
        }
        if (zoneId == 551){
            return 1;
        }
        if (zoneId == 495){
            return 2;
        }
    }
    
    if (speciesId == 550){
        if (zoneId == ABUNDANT){
            return 1;
        }
    }

    if (speciesId == 555){
        if (zoneId == 181 || zoneId == ABUNDANT){ // ChANGE THIS LATER
            return 1;
        }
    }
    return 0;
}




static int sub_GenerateWildPokemonForZoneID(int zoneId, int level, int isWalking, int isShaking, int isDoubles, int isWater) {
    int listMenu; 
    int yesNo; 
    int wildOutcome; 

Label_RerollingEncountersWildTables: ;
    int speciesId;
    int forme; 
    int level; 
    // The options for encounters in [VAR LOCATION(1)] are:
    Message.System(textId + 19, 2);

    ListMenu.InitRightAnchor(31, 5, 0, 1, listMenu);
    if (isWalking == 1){
        ListMenu.AddItem(textId+20, 65535, 0); // Grass / Walking
    }
    if (isWalking == 2){
        ListMenu.AddItem(textId+21, 65535, 0); // Ground
    }
    if (isShaking == 1){
        if (TrainerCard.HasBadge(0)){
            ListMenu.AddItem(textId+22, 65535, 1); // Shaking Grass 
        }
    }
    if (isShaking == 2){
        ListMenu.AddItem(textId+23, 65535, 1); // DustClouds 
    }
    if (isShaking == 3){
        ListMenu.AddItem(textId+24, 65535, 1); // Shadows 
    }
    if (isDoubles == 1){
        ListMenu.AddItem(textId+25, 65535, 6); // Doubles Grass
    }
    if (isFishing == 1){
        ListMenu.AddItem(textId+26, 65535, 2); // Fishing
        if (TrainerCard.HasBadge(0)){
            ListMenu.AddItem(textId+27, 65535, 3); // Fishing Dark Spot
        }
    }
    if (isSurfing == 1){
        if (TrainerCard.HasBadge(4)){
            ListMenu.AddItem(textId+28, 65535, 4); // Surfing
            ListMenu.AddItem(textId+29, 65535, 5); // Surfing Dark Spot
        }
    }
    ListMenu.AddItem(textId+6, 65535, 7); // Exit 
    ListMenu.ShowDialog();


    // RANDOMLY ROLL THE ENCOUNTER 
    if (listMenu >= 0 && listMenu <= 6){
        speciesId = GenerateWildsEncounter(435, listMenu);
    }
    else {
        Message.CloseAll();
        return 0; 
    }


Label_WildEncounterDialoguePopup: ;
    WordSet.LoadPokeSpecies(4, speciesId);
    // Your encounter is [VAR PKNAME(4)]!\c\nIs this a valid encounter?
    Message.System(textId + 20, 2);

    yesNo = YesNoWin.Show();
    if (yesNo == 0){
        forme = getForme(speciesId, zoneId);
        wildOutcome = callWildBattle(speciesId, level, forme); 
    } else {
        goto Label_RerollingEncountersWildTables; 
    }

    if (wildOutcome == 0){
        goto Label_WildEncounterDialoguePopup;
    } else {
        Message.System(textId + 18, 2);
        yesNo = YesNoWin.Show();
        if (yesNo == 0){
            return 0;
        } else {
            return 1;
        }
    }
}
