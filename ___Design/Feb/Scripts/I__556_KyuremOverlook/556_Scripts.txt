	static void sub_CustomizeRescue(){
		int listMenu_Natures; 
		int listMenu_Abilities; 
		int yesNo; 
		int offset;
		int abilCurrent;
		int abil1; 
		int abil2; 
		int abil3; 
		int naturePage = 0; 

		LABEL_ChooseNature:  
		Message.Info(160, 2);
		ListMenu.InitRightAnchor(31, 5, 0, 1, listMenu_Natures);
		offset = naturePage * 5;
		ListMenu.AddItem((163 + offset), 65535, (1 + offset)); // Hardy
		ListMenu.AddItem((164 + offset), 65535, (2 + offset)); // Lonely
		ListMenu.AddItem((165 + offset), 65535, (3 + offset)); // Brave
		ListMenu.AddItem((166 + offset), 65535, (4 + offset)); // Adamant 
		ListMenu.AddItem((167 + offset), 65535, (5 + offset)); // Naughty 
		if (naturePage != 4){
			ListMenu.AddItem(161, 65535, 30); // Next Page 
		}
		if (naturePage != 0){
			ListMenu.AddItem(162, 65535, 31);
		}
		ListMenu.ShowDialog();
		
		if (listMenu_Natures == 30){
			naturePage = naturePage + 1;
			goto LABEL_ChooseNature;
		}
		else if (listMenu_Natures == 31){
			naturePage = naturePage - 1; 
			goto LABEL_ChooseNature;
		}
		else {
			WordSet.LoadNature(7, listMenu_Natures);
			Message.Info(192, 2);
			yesNo = YesNoWin.Show();
			if (yesNo == 1){
				Message.CloseAll();
				goto LABEL_ChooseNature;
			}
			Message.Info(199, 2);
		}

		LABEL_ChooseAbilities: 
		abil1 = DayCare.GetPkmGrowLevel(0);
		abil2 = DayCare.GetPkmGrowLevel(1);
		abil3 = DayCare.GetPkmGrowLevel(2);
		abilCurrent = DayCare.GetPkmGrowLevel(7);
		WordSet.LoadAbility(7, abilCurrent);
		WordSet.LoadAbility(8, abil1);
		WordSet.LoadAbility(9, abil2);
		WordSet.LoadAbility(10, abil3);
		
		Message.Info(193, 2);
		ListMenu.InitRightAnchor(31, 5, 0, 1, listMenu_Abilities);
		if (abil1 != 0){
			ListMenu.AddItem(194, 65535, 4);
		}
		if (abil2 != 0){
			ListMenu.AddItem(195, 65535, 5);
		}
		if (abil3 != 0){
			ListMenu.AddItem(196, 65535, 6);
		}
		ListMenu.AddItem(197, 65535, 8); // No Change
		ListMenu.ShowDialog();

		if (listMenu_Abilities == 8){
			WordSet.LoadAbility(11, abilCurrent);
		}
		else if (listMenu_Abilities == 4){
			WordSet.LoadAbility(11, abil1);
		}
		else if (listMenu_Abilities == 5){
			WordSet.LoadAbility(11, abil2);
		}
		else if (listMenu_Abilities == 6){
			WordSet.LoadAbility(11, abil3);
		}
		else {

		}
		Message.Info(198, 2);
		yesNo = YesNoWin.Show(); 
		if (yesNo == 1){
			goto LABEL_ChooseAbilities;
		}
		else { 
			abilCurrent = DayCare.GetPkmGrowLevel(listMenu_Abilities);
			WordSet.LoadAbility(11, abilCurrent);
			Message.Info(200, 2);
			Message.CloseAll();
		}
	}
