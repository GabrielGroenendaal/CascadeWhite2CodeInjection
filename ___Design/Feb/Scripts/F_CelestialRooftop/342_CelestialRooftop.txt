	// Ringing the Gong 
	static void sub_GenerateWildPokemon() {
		int random; 
		int speciesId;
		int form;
		int level;
		int isVictory;
		if (EventFlags.Get(1520 + 769)) {
			Actor.PauseAll();
			random = System.Random(100);
			level = System.Random(2) + 39;
			form = 0;
			if (random < 100) {
				if (random < 75) {
					if (random < 50) {
						if (random < 40){
							if (random < 30) {
								if (random < 20) {
									if (random < 10) {
										speciesId = 356; // Togetic
									}
									else {
										speciesId = 593; // Misdreavus
									}
								}
								else {
									speciesId = 604; // Noctowl
								}
							}
							else {
								speciesId = 367; // Murkrow
							}
						}
						else {
							speciesId = 368; // Altaria
						}
					}
					else {
						speciesId = 369; // Castform
						form = System.Random(3);
					}
				}
				else {
					speciesId = 358; // Chimecho
				}
			} 
			else {
				speciesId = 358; // Chimecho
			}

			Sound.PVPlay(speciesId, 0);
			Sound.PVWait();
			Battle.CallWildBattleEx(speciesId, level, form,  0);	
			isVictory = Battle.IsWildVictory();
			if (isVictory != 1){ 
				Battle.CallWildLose();
				goto Label_LostWildBattle;		
			}
			Battle.EndWildBattle();
			
			Label_LostWildBattle:
			Runtime.FinishSubEvents();
			Actor.UnpauseAll();
		}
	}
