    public static void main_1(){
        
        int playerX;
        int playerZ;

        int GRUNTA_ENTITY = 1;
        int GRUNTB_ENTITY = 2;
        int isVictory;

        Actor.PauseAll();
        
        Message.ActorEx(1024, 1, GRUNTA_ENTITY, 4, 0);
        Message.CloseAll();
		

        Fade.OutBlackQ();
		Fade.Wait();
        EventFlags.Clear(574);
        Runtime.Sleep(16);
        Actor.Show(GRUNTB_ENTITY);
        Fade.InBlackQ();
		Fade.Wait();

        ShowExclamationMark(255);
        TurnNorth8F(255);
        Actor.WaitActionFinish();

        Sound.BGMPlay(1127);
        Message.ActorEx(1024, 2, GRUNTA_ENTITY, 4, 1);
        Message.CloseAll();
        GruntRushesOneTileDown(GRUNTB_ENTITY);
        Actor.WaitActionFinish();
		goto SKIP_BATTLE;
        Battle.CallTrainerBattle(434, 0, 0);
		isVictory = Battle.IsTrainerVictory();
		if (isVictory != 1) {
			Battle.CallTrainerLose();
            goto Label_EndGruntScript;
		}           
        Battle.EndTrainerBattle();

		SKIP_BATTLE:
        GruntBacksTwoTilesUp(GRUNTB_ENTITY);
        Message.ActorEx(1024, 3, GRUNTB_ENTITY, 2, 0);
        Actor.WaitActionFinish();
        Message.CloseAll();
        
        HopUp(GRUNTA_ENTITY);
        TurnSouth8F(255);
        Message.ActorEx(1024, 4, GRUNTA_ENTITY, 4, 1);
        Actor.WaitActionFinish();
        Message.CloseAll();

		goto SKIP_BATTLE2;
        Battle.CallTrainerBattle(443, 0, 0);
		isVictory = Battle.IsTrainerVictory();
		if (isVictory != 1) {
			Battle.CallTrainerLose();
            EventFlags.Set(574);
            goto Label_EndGruntScript;
		}           
        Battle.EndTrainerBattle();
		SKIP_BATTLE2:
        Field.PlayMapBGM();
        Message.ActorEx(1024, 5, GRUNTA_ENTITY, 4, 0);
        Actor.WaitActionFinish();
        Message.CloseAll();

        EventFlags.Set(350);
        EventFlags.Set(574);
        EventWorks.Set(16716, 1);

        Label_EndGruntScript:
        Runtime.FinishSubEvents();
		Actor.UnpauseAll();
    }
