	public static void main_5() {

		WordSet.LoadRivalName(1);

			WordSet.LoadPartyPokeName(3, name);

			WordSet.LoadDaycarePokeSpecies(3, 0);
		}
		if (EventWorks.Get(16512)==5){
			sub_ZinzolinEncounter();
		}
		else {
			Sound.SEPlay(1351);
			Actor.FacePlayer();
			if (EventWorks.Get(16512) == 3){
				Message.Actor(1024, 70, 2, 0);
			}
			else {
				if (PokeParty.GetPkmCount(0) == 6) {
					messageId = 72;
				}
				else {
					messageId = 71;
				}
				Message.Actor(1024, messageId, 2, 0);
			}
			Message.CloseActor();
			Runtime.FinishSubEvents();
		}
	}
