    // Back to Back Trigger 
    public static void main_1(){
        
        int playerX;
        int playerZ;

        int GRUNTA_ENTITY = 0;
        int GRUNTB_ENTITY = 1;
        int isVictory;

        Actor.PauseAll();

        ShowExclamationMark(GRUNTA_ENTITY);
        Actor.WaitActionFinish();

        Sound.BGMPlay(1127);
        TurnNorth8F(255);
        WalkSouthOneTile(GRUNTA_ENTITY);
        Actor.WaitActionFinish();

        Message.ActorEx(1024, 0, GRUNTA_ENTITY, 4, 0);
        Message.CloseAll();

		//goto skipbattle;
        Battle.CallTrainerBattle(430, 0, 0);
		isVictory = Battle.IsTrainerVictory();
		if (isVictory != 1) {
			Battle.CallTrainerLose();
            goto Label_EndGruntScript;
		}           
        Battle.EndTrainerBattle();

		skipbattle:
		Field.PlayMapBGM();
        GruntBacksOneTileUp(GRUNTA_ENTITY);
        Message.ActorEx(1024, 1, GRUNTA_ENTITY, 4, 0);
        Actor.WaitActionFinish();
        Message.CloseAll();
        
        Sound.SEPlay(2222);
		Sound.SEWait();
		Message.Info(2, 2);
		Message.CloseInfo();

        Sound.SEPlay(1369);
        EventFlags.Clear(574);
        Actor.Show(GRUNTB_ENTITY);
		Sound.SEWait();
        WalkEastOneTile(GRUNTB_ENTITY);
        Actor.WaitActionFinish();


        ShowExclamationMark(GRUNTB_ENTITY);
        Actor.WaitActionFinish();
        Sound.BGMPlay(1127);
        Player.GetGPos(playerX, playerZ);

        Actor.WalkRoute(GRUNTB_ENTITY, playerX-1, playerZ, 0, 4, 0);
        Runtime.Sleep(8);
        TurnWest8F(255);
        Actor.WaitActionFinish();
        Message.ActorEx(1024, 4, GRUNTB_ENTITY, 4, 0);
        Message.CloseAll();

		//goto skipbattle2;
        Battle.CallTrainerBattle(441, 0, 0);
		isVictory = Battle.IsTrainerVictory();
		if (isVictory != 1) {
			Battle.CallTrainerLose();
            EventFlags.Set(574);
            goto Label_EndGruntScript;
		}           
        Battle.EndTrainerBattle();

		skipbattle2:
		Field.PlayMapBGM();
        GruntBacksFourTilesLeft(GRUNTB_ENTITY);
		Actor.WaitActionFinish();
        Message.ActorEx(1024, 5, GRUNTB_ENTITY, 2, 0);
        Actor.WaitActionFinish();
        Message.CloseAll();

        TurnNorth8F(255);
        Message.ActorEx(1024, 6, GRUNTA_ENTITY, 4, 0);
        Actor.WaitActionFinish();
        Message.CloseAll();

        EventFlags.Set(348);
        EventFlags.Set(574);
        EventWorks.Set(16714, 1);

        Label_EndGruntScript:
        Runtime.FinishSubEvents();
		Actor.UnpauseAll();
    }
