    // This script checks to see what the lowest health non-fainted Pokemmon in the party is and takes it. 
    static void stealPokemon() {
        // this script is going to iterate over the players team and steal one of their pokemon. 
        int lowestHealthPokemon = 12;
        int lowestHealth = 999;
        int totalPokemonCount = PokeParty.GetPkmCount(0);
        int i;
        int monsHealth;
        for (i = 0; i < totalPokemonCount; i++){
            if (!PokeParty.IsEgg(i)){
                monsHealth = PokeParty.GetParam(i, 160);
                if (monsHealth > 0){
                    if (monsHealth < lowestHealth){
                        lowestHealth = monsHealth;
                        lowestHealthPokemon = i;
                    }
                    if (monsHealth == lowestHealth){
                        if (System.Random(100) > 50){
                            lowestHealth = monsHealth;
                            lowestHealthPokemon = i;
                        }
                    }
                }
            }
        }
        if (lowestHealthPokemon == 12){
            stealPokemonCheck2();
        }
        else {
            stealThePokemon(lowestHealthPokemon);
        }
        return;
    }

        // this script is going to iterate over the players team and steal one of their pokemon. 
        int option1; 
        int option2;
        int option3;
        int option4;
        int option5;
        int option6;
        
        int viablePokemonCount;
        int totalPokemonCount;
        int i;
        int currentHealth;
        int dummy;
        int random;
        viablePokemonCount = 0;

        dummy = PokeParty.GetEVTotal(0); // This heals all the non-fainted Pokemon in the party.
        totalPokemonCount = PokeParty.GetPkmCount(0);

        for (i = 0; i < totalPokemonCount; i++){
            if (!PokeParty.IsEgg(i) && PokeParty.IsFullHP(i)){
                if (viablePokemonCount == 0) {
                	option1 = i;
                }
                if (viablePokemonCount == 1) {
                	option2 = i;
                }
                if (viablePokemonCount == 2) {
                	option3 = i;
                }
                if (viablePokemonCount == 3) {
                	option4 = i;
                }
                if (viablePokemonCount == 4) {
                	option5 = i;
                }
                if (viablePokemonCount == 5) {
                	option6 = i;
                }
                viablePokemonCount = viablePokemonCount + 1;
            }
        }

        if (viablePokemonCount == 0) {
            PokeParty.RecoverAll();
            random = PokeParty.GetMemberByType(2);
            Daycare.DepositPkm(random); 
            goto LABEL_EndSteal;
        }
        
        if (viablePokemonCount == 1) {
            random = option1;
            PokeParty.RecoverAll();
            Daycare.DepositPkm(random);
            goto LABEL_EndSteal;
        }

        random = System.Random(viablePokemonCount - 1);

        if (random == 0) {
        	random == option1;
        }
         if (random == 1) {
        	random == option2;
        }
         if (random == 2) {
        	random == option3;
        }
         if (random == 3) {
        	random == option4;
        }
         if (random == 4) {
        	random == option5;
        }
         if (random == 5) {
        	random == option6;
        }

        LABEL_EndSteal:
        return;
