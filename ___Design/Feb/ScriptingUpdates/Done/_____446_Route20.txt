	static void repositionJanine(int janineEntity, int playerX, int playerZ, int janineX, int janineZ){
		LockDirection(janineEntity);
		ShowExclamationMark(janineEntity);

		if (playerX > janineX){
			WalkWest(janineEntity);
			goto Label_PostJanineMovement;
		}
		if (playerX < janineX){
			WalkEast(janineEntity);
			goto Label_PostJanineMovement;
		}
		if (playerZ > janineZ){
			WalkNorth(janineEntity);
			goto Label_PostJanineMovement;
		}
		if (playerZ < janineZ){
			WalkSouth(janineEntity);
			goto Label_PostJanineMovement;
		}
		Label_PostJanineMovement:
		UnlockDirection(janineEntity);
	}
