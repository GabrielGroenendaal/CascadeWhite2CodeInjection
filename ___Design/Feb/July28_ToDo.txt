    extern "C" void HandlerCollective(BattleEventItem * item, ServerFlow * serverFlow, u32 pokemonSlot, u32 * work) {
        if (pokemonSlot == BattleEventVar_GetValue(VAR_MON_ID)) {
            BattleMon* currentMon = Handler_GetBattleMon(serverFlow, pokemonSlot);

            //  0: Combee-Male 
            //  1: Combee-Female 
            //  2: Combee-Collective
            // 26: Unown-Exclamation 
            // 27: Unown-Question 
            // 28: Unown-Exclamation Schooling 
            // 29: Unown-Question Schooling

            u8 currentForm = BattleMon_GetValue(currentMon, VALUE_FORM);
            u8 newForm = currentForm;
            u16 msgID = BATTLE_SCHOOLING_INACTIVE_MSGID;

            // TODO: ADD WISHIWASHI
            if (BattleMon_GetSpecies(currentMon) == 201) {
                u32 currentHP = BattleMon_GetValue(currentMon, VALUE_CURRENT_HP);
                if (currentForm == 26 || currentForm == 27){
                    if (currentHP > DivideMaxHPZeroCheck(currentMon, 4u)) {
                        newForm = currentForm - 2;
                        msgID = BATTLE_SCHOOLING_ACTIVE_MSGID;
                    }
                }
                if (currentForm == 28 || currentForm == 29){
                    if (currentHP <= DivideMaxHPZeroCheck(currentMon, 4u)) {
                        newForm = currentForm + 2;
                        msgID = BATTLE_SCHOOLING_INACTIVE_MSGID;
                    }
                } 
            }

            if (BattleMon_GetSpecies(currentMon) == 415) {
                u32 currentHP = BattleMon_GetValue(currentMon, VALUE_CURRENT_HP);
                if (currentForm == 0){
                    if (currentHP > DivideMaxHPZeroCheck(currentMon, 4u)) {
                        newForm = currentForm + 2;
                        msgID = BATTLE_SCHOOLING_ACTIVE_MSGID;
                    }
                }
                if (currentForm == 2){
                    if (currentHP <= DivideMaxHPZeroCheck(currentMon, 4u)) {
                        newForm = currentForm - 2;
                        msgID = BATTLE_SCHOOLING_INACTIVE_MSGID;
                    }
                } 
            }


            if (currentForm != newForm) {
                    HandlerParam_ChangeForm* changeForm;
                    changeForm = (HandlerParam_ChangeForm*)BattleHandler_PushWork(serverFlow, EFFECT_CHANGE_FORM, pokemonSlot);
                    changeForm->header.flags |= HANDLER_ABILITY_POPUP_FLAG;
                    changeForm->pokeID = pokemonSlot;
                    changeForm->newForm = newForm;
                    BattleHandler_StrSetup(&changeForm->exStr, 2u, msgID);
                    BattleHandler_PopWork(serverFlow, changeForm);
                }

        }
    }
    BattleEventHandlerTableEntry CollectiveHandlers[]{
        {EVENT_SWITCH_IN, HandlerCollective},
        {EVENT_TURN_CHECK_DONE, HandlerCollective},
    };
    extern "C" BattleEventHandlerTableEntry * EventAddZenMode(u32 * handlerAmount) {
        *handlerAmount = ARRAY_COUNT(SchoolingHandlers);
        return CollectiveHandlers;
    }
