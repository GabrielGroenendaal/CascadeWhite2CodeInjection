                                if (atkAbility == ABIL105_MOISTURIZE && Type == TYPE_NORMAL)
                                {
                                    Type = TYPE_WATER;

                                {
                                    v10 = BasePower << 18;
                                }
                                else
                                {
                                }

                                {

#if SWITCH_AI_CHANGES

                                }

                                if (v23[i] < BasePower)
                                {
                                    v23[i] = BasePower;
                                }
                            }
                        }
                        v5 = (v5 + 1);
                    } while (v5 < MoveCount);
                }
            }
        }
        result = a3;
        for (j = 0; j < a3; result = a3)
        {
            for (k = (j + 1); k < a3; k = (k + 1))

                v14 = v23[k];

        }
