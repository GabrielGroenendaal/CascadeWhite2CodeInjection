    static int sub_GenerateWildPokemonForZoneID(int zoneId, int level, int isWalking, int isShaking, int isDoubles, int isWater) {
        int listMenu; 
        int listMenu2;
        int yesNo; 
        int numberOfTables = 0;
        int textId = 9;
        int random; 

        Label_RePickingEncounterWildTables: ;
        int speciesId;
        int forme; 

        if (EventWorks.Get(16567) > 0){
            
            int encountType;
            int encountNum = EventWorks.Get(16504);
            int postCheren = (encountNum > 9) ? 1 : 0; 
            int postClay = (encountNum > 35) ? 1 : 0;

            Label_RandomizeEncounterTableWild:
            int random = System.Random(100);

            
            // 30%: Walking 
            // Walking Exists: Walking 
            // Walking Does Not Exist and Doubles Exists: Doubles 
            // Walking / Doubles does not exist: Fishing Regular 
            if (random < 30){
                  encountType = (isWalking > 0) ? 0 : ((isDoubles > 0) ? 6 : 2);
            }

            // 20% Shaking Grass 
            // Shaking Exists and Post Cheren: Shaking Grass 
            // Shaking Exists and Pre Cheren: Regular Grass 
            // Shaking Does Not Exist and Doubles Exists: Doubles 
            // Shaking/Doubles Does not Exist: Fishing, Dark Spot
            else if (random < 50){
                  encountType = (isShaking > 0) ? ((postCheren == 1) ? 1 : 0) : ((isDoubles > 0) ? 6 : 3);
            }
            // 20% Doubles Grass 
            // Doubles Exists: Doubles 
            // Doubles Does not Exist but Walking Exists: Walking 
            // Doubles / Walking Does not Exist, Pre-Clay, Pre-Cheren: Fishing Regular 
            // Doubles / Walking Does Not Exist, Pre-Clay: Fishing, Dark Spot 
            // Doubles / Walking Does Not Exist: Surf Regular 
            else if (random < 70){
                  encountType = (isDoubles > 0) ? 6 : ((isWalking > 0) ? 0 : ((postClay == 1) ? 4 : ((postCheren == 1) ? 3 : 2)));
            }
            // 8% Fishing 
            // Water Exists and Pre Clay: Fishing 
            // Water Exists and Post Clay: Surfing Dark Spot 
            // Water Does Not Exist and Walking Exists: Walking 
            // Water / Walking Does not Exist: Doubles 
            else if (random < 78) {
                  encountType = (isWater > 0) ? 2 : ((isWalking > 0) ? 0 : 6);
            }
            // 7% Fishing Dark Spot
            // Water Exists and Post Cheren: Fishing Dark Spot 
            // Water Exists and Pre Cheren: Fishing Regular 
            // Water Does Not Exist and Shaking Exists: Walking 
            // Water / Shaking does not exist and Walking Exists: Walking
            // Water/Shaking/Walking does not Exist: Doubles 
            else if (random < 85){
                  encountType = (isWater > 0) ? ((postCheren == 1) ? 3 : 2) : ((isShaking > 0) : 0 : ((isWalking > 0) ? 0 : 6));
            }

            // 8% Surfing
            // Water Exists and Post Clay: Surfing 
            // Water Exists and Pre Clay: Fishing, Dark Spot 
             // Water Exists and Pre Clay, Pre Cheren: Fishing
            // Water Does Not Exist and Shaking Exists: Shaking 
            // Water / Shaking Does not Exist and Walking Exists: Walking 
            // Water / Shaking / Walking does not Exist: Doubles 
            else if (random < 93){
                  if (isWater > 0){
                        if (postClay == 1){
                              encountType = (isWalking > 0) ? 4 : 5;
                        }
                        else {
                              if (postCheren == 1){
                                    encountType = (isWalking > 0) ? 3 : 2;
                              }
                              else {
                                    encountType = 2;
                              }
                        }
                  }
                  else {
                        if (isShaking > 0){
                              encountType = 1;
                        }
                        else {
                              if (isDoubles > 0){
                                    encountType = 6;
                              }
                              else {
                                    encountType = 0;
                              }
                        }
                  }
            }

            // 7% Surfing Dark Spot
            // Water Exists and Post Clay: Surfing Dark Spot
            // Water Exists and Pre Clay: Fishing
            // Water Exists and Pre Clay, Pre Cheren: Fishing
            // Water Does Not Exist and Shaking Exists: Shaking 
            // Water / Shaking Does not Exist and Walking Exists: Walking 
            // Water / Shaking / Walking does not Exist: Doubles 
            else if (random < 100){
                  encountType = (isWater > 0) ? ((postClay == 1) ? 5 : 2) : ((isShaking > 0) ? 1 : ((isDoubles > 0) ? 6 : 0));
            }
            else {
                  goto Label_RandomizeEncounterTableWild;
            }

            speciesId = GenerateWildsEncounter(zoneId, encountType);
            callWildBattle(speciesId, level, 0, 0); 

            if (EventWorks.Get(16567) == 1){
                  WordSet.LoadPokeSpecies(4, speciesId);
                  forme = getForme(speciesId, zoneId);
                  FieldDemo.CallPokemonPreview(speciesId, forme, 0, 0);
                  // Your encounter is [VAR PKNAME(4)]!
                  Message.System(135, 2);
                  Runtime.Sleep(48);
                  Message.CloseAll();
            }
            return 0;
        }

        // The options for encounters in [VAR LOCATION(1)] are:
        Message.System(28, 2);

        ListMenu.InitRightAnchor(31, 5, 0, 1, listMenu);
        if (isWalking > 0){
            numberOfTables += 1;
        }
        if (isWalking == 1){
                ListMenu.AddItem(29, 65535, 0); // Grass / Walking
        }
        if (isWalking == 2){
                ListMenu.AddItem(30, 65535, 0); // Ground
        }
        if (isWalking == 4){
                ListMenu.AddItem(39, 65535, 0); // Sand, Exterior
        }


        if (isShaking > 0){
            if (TrainerCard.HasBadge(0)){
                numberOfTables = numberOfTables + 1;

                if (isShaking == 1){
                    ListMenu.AddItem(31, 65535, 1); // Shaking Grass 
                }
                if (isShaking == 2){
                    ListMenu.AddItem(32, 65535, 1); // DustClouds 
                }
                if (isShaking == 3){
                    ListMenu.AddItem(33, 65535, 1); // Shadows 
                }
                if (isShaking == 4){
                    ListMenu.AddItem(40, 65535, 1); // Sand, Interior 
                }
            }
        }
       

        if (isDoubles == 1){
            ListMenu.AddItem(34, 65535, 6); // Doubles Grass
            numberOfTables += 1;
        }
        if (isWater > 0){
            ListMenu.AddItem(35, 65535, 2); // Fishing
            numberOfTables += 1;
            if (TrainerCard.HasBadge(0)&& isWater != 2){
                ListMenu.AddItem(36, 65535, 3); // Fishing Dark Spot
                numberOfTables += 1;
            }
            if (TrainerCard.HasBadge(4)){
                ListMenu.AddItem(37, 65535, 4); // Surfing
                numberOfTables += 1;
                if (isWater != 2){ 
                        ListMenu.AddItem(38, 65535, 5); // Surfing Dark Spot
                        numberOfTables += 1;
                }
            }
        }

        ListMenu.AddItem(15, 65535, 7); // Exit 
        ListMenu.ShowDialog();

        Label_RerollingEncountersWildTables: ; 
        // RANDOMLY ROLL THE ENCOUNTER 
        if (listMenu >= 0 && listMenu <= 6){
                speciesId = GenerateWildsEncounter(zoneId, listMenu);
        }
        else {
                Message.CloseAll();
                return 1; 
        }


        Label_WildEncounterDialoguePopup: ;
        WordSet.LoadPokeSpecies(4, speciesId);
        forme = getForme(speciesId, zoneId);
        FieldDemo.CallPokemonPreview(speciesId, forme, 0, 0);
        // Your encounter is [VAR PKNAME(4)]!\c\nDo you want to move forward with\n[VAR PKNAME(4)] as your encounter?
        Message.System(textId + 32, 2);


        ListMenu.InitRightAnchor(31, 5, 0, 1, listMenu2);
        ListMenu.AddItem(49, 65535, 0); // Continue with Encounter #[VAR NUM(0)]
        ListMenu.AddItem(50, 65535, 1); // Reroll on the same table
        if (numberOfTables > 1){
            ListMenu.AddItem(51, 65535, 2); // Reroll on a different table
        }
        ListMenu.AddItem(15, 65535, 3); // Exit 
        ListMenu.ShowDialog();

        if (listMenu2 == 0){
            random = System.Random(2);
            if (listMenu == 2 || listMenu == 3){
                callWildBattle(speciesId, (10+random), forme, 2); 
            }
            else if (listMenu == 4 || listMenu == 5){
                callWildBattle(speciesId, (30+random), forme, 2); 
            }
            else {
                callWildBattle(speciesId, (level+random), forme, 2); 
            }
        }
        else if (listMenu2 == 1){
            goto Label_RerollingEncountersWildTables; 
        }
        else if (listMenu2 == 2){
            goto Label_RePickingEncounterWildTables; 
        }
        else {
            return 1; 
        }

        Message.System(textId + 18, 2);
        yesNo = YesNoWin.Show();
        if (yesNo == 0){
            return 0;
        } else {
            return 2;
        }
        return 2;
    }
